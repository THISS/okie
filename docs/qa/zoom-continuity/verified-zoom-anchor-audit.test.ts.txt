import {readFileSync,writeFileSync} from 'node:fs';
import {it} from 'vitest';
import {compileScanFixture} from './src/renderer/scanFixture';
import {semanticBounds} from './src/renderer/goldenC4Scene';
it('compares actual cold and paged component anchors',()=>{
const snapshot=JSON.parse(readFileSync('/tmp/zoom-coverage-snapshot.json','utf8')),view=JSON.parse(readFileSync('/tmp/zoom-coverage-view.json','utf8')),story=JSON.parse(readFileSync('../../fixtures/scan/thiss__okie/story.json','utf8'));
const fixture=compileScanFixture({snapshot,view,story},{targetAspect:1.6}); const web=fixture.createScene('container:apps-web');
let checkedRepresentations=0;const peerChecks:any[]=[];const protocolMismatches:any[]=[];const verify=(scene:any,label:string)=>{for(const e of snapshot.entities.filter((e:any)=>e.kind==='externalSystem'||e.kind==='person')){const expected=web.projection!.boundsByEntityIdAndDetail[e.id]?.context;const actual=scene.projection.boundsByEntityIdAndDetail[e.id];peerChecks.push({label,id:e.id,bounds:actual,matches:Object.values(actual??{}).every(b=>JSON.stringify(b)===JSON.stringify(expected))});}for(const object of scene.protocolSnapshot.objects){const id=scene.projection.visualToSemanticEntityId[object.id];if(!id)continue;for(const rep of object.representations){const detail=rep.id.split(':').at(-1);const expected=scene.projection.boundsByEntityIdAndDetail[id]?.[detail];if(!expected)continue;checkedRepresentations++;if(['x','y','width','height'].some(k=>Math.abs(expected[k]-rep.bounds[k])>1e-9))protocolMismatches.push({label,id,detail,expected,actual:rep.bounds});}}};verify(web,'web');const rows=['component:apps-web-src-hosted-atlas-ts','component:apps-web-src-renderer-fixture-bundle-ts','component:apps-web-src-inspector-scan-one-pager-ts'].map(id=>{
const source=semanticBounds(web,id,'component'); const cold=fixture.createScene(id);const live=fixture.createScene(id,web); const center=(r:any)=>r?{x:r.x+r.width/2,y:r.y+r.height/2}:null;const c=center(source)!;
const codeCenter=center(semanticBounds(cold,id,'code'))!;const window={x:codeCenter.x-20,y:codeCenter.y-20,width:40,height:40};const keepId=snapshot.entities.find((e:any)=>e.parentId===id)?.id;const paged=fixture.createScene(id,cold,{worldBounds:window,keepEntityIds:[id,keepId]});
const coldBounds=cold.projection!.boundsByEntityIdAndDetail;const pagedBounds=paged.projection!.boundsByEntityIdAndDetail;const codeIds=cold.projection!.entityIdsByDetail.code.filter(cid=>snapshot.entities.find((e:any)=>e.id===cid)?.kind==='code');const pagedCodeIds=paged.projection!.entityIdsByDetail.code.filter(cid=>snapshot.entities.find((e:any)=>e.id===cid)?.kind==='code');const intersect=(r:any)=>r&&r.x<window.x+window.width&&r.x+r.width>window.x&&r.y<window.y+window.height&&r.y+r.height>window.y;const expectedInWindow=codeIds.filter(cid=>intersect(coldBounds[cid]?.code));const missingInWindow=expectedInWindow.filter(cid=>!pagedCodeIds.includes(cid));const sharedChanges=pagedCodeIds.filter(cid=>codeIds.includes(cid)&&JSON.stringify(coldBounds[cid]?.code)!==JSON.stringify(pagedBounds[cid]?.code));verify(cold,id+':cold');verify(live,id+':live');verify(paged,id+':paged');return {id,coldLiveCodeDifferences:codeIds.filter(cid=>JSON.stringify(coldBounds[cid]?.code)!==JSON.stringify(live.projection!.boundsByEntityIdAndDetail[cid]?.code)),coldLiveCodeMembershipEqual:JSON.stringify(cold.projection!.entityIdsByDetail.code)===JSON.stringify(live.projection!.entityIdsByDetail.code),expectedInWindow,missingInWindow,sharedChanges,source,sourceCenter:c,cold:semanticBounds(cold,id,'component'),live:semanticBounds(live,id,'component'),paged:semanticBounds(paged,id,'component'),coldCode:semanticBounds(cold,id,'code'),pagedCode:semanticBounds(paged,id,'code'),window,coldComponentIds:cold.projection?.entityIdsByDetail.component.length,pagedComponentIds:paged.projection?.entityIdsByDetail.component.length,coldCodeIds:cold.projection?.entityIdsByDetail.code.length,pagedCodeIds:paged.projection?.entityIdsByDetail.code.length};});
writeFileSync('../../docs/qa/zoom-continuity/verified-component-anchor-audit.json',JSON.stringify({snapshotId:snapshot.id,targetAspect:1.6,checkedRepresentations,protocolMismatches,peerChecks,rows},null,2)+'\n');console.log(JSON.stringify(rows));
},60000);
