import {readFileSync,writeFileSync} from 'node:fs';import {createHash} from 'node:crypto';import {it} from 'vitest';
import {compileScanFixture} from './src/renderer/scanFixture';import {semanticBounds} from './src/renderer/goldenC4Scene';
it('audits shallow owner displacement with current built compiler',()=>{
const snapshot=JSON.parse(readFileSync('/tmp/zoom-coverage-snapshot.json','utf8')),view=JSON.parse(readFileSync('/tmp/zoom-coverage-view.json','utf8')),story=JSON.parse(readFileSync('../../fixtures/scan/thiss__okie/story.json','utf8'));const fixture=compileScanFixture({snapshot,view,story},{targetAspect:1.6});const root=fixture.createScene(view.rootEntityId);const center=(r:any)=>({x:r.x+r.width/2,y:r.y+r.height/2});const compare=(from:any,to:any)=>{if(!from||!to)return null;const a=center(from),b=center(to);return{from,to,delta:{x:b.x-a.x,y:b.y-a.y,distance:Math.hypot(b.x-a.x,b.y-a.y)}}};const rows=snapshot.entities.filter((e:any)=>e.kind==='container').map((e:any)=>{const target=fixture.createScene(e.id);return{id:e.id,l2ToL3:compare(semanticBounds(root,e.id,'container'),semanticBounds(target,e.id,'component')),withinRootL2L3:compare(semanticBounds(root,e.id,'container'),semanticBounds(root,e.id,'component')),l2AcrossFocus:compare(semanticBounds(root,e.id,'container'),semanticBounds(target,e.id,'container'))};});const result={snapshotId:snapshot.id,targetAspect:1.6,compilerDistSha256:createHash('sha256').update(readFileSync('../../packages/scene-compiler/dist/compile-c4.js')).digest('hex'),architectureDistSha256:createHash('sha256').update(readFileSync('../../packages/architecture/dist/c4.js')).digest('hex'),l1ToL2:compare(semanticBounds(root,view.rootEntityId,'context'),semanticBounds(root,view.rootEntityId,'container')),rows};writeFileSync('../../docs/qa/zoom-continuity/final-shallow-owner-centers.json',JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify({l1ToL2:result.l1ToL2,rows:rows.map(r=>({id:r.id,l2ToL3:r.l2ToL3?.delta,withinRoot:r.withinRootL2L3?.delta,acrossFocus:r.l2AcrossFocus?.delta}))}));
},60000);
