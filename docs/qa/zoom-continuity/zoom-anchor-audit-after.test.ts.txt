import {readFileSync,writeFileSync} from 'node:fs';
import {it} from 'vitest';
import {compileScanFixture} from './src/renderer/scanFixture';
import {semanticBounds} from './src/renderer/goldenC4Scene';
it('compares actual cold and paged component anchors',()=>{
const snapshot=JSON.parse(readFileSync('/tmp/zoom-coverage-snapshot.json','utf8')),view=JSON.parse(readFileSync('/tmp/zoom-coverage-view.json','utf8')),story=JSON.parse(readFileSync('../../fixtures/scan/thiss__okie/story.json','utf8'));
const fixture=compileScanFixture({snapshot,view,story},{targetAspect:1.6}); const web=fixture.createScene('container:apps-web');
const rows=['component:apps-web-src-hosted-atlas-ts','component:apps-web-src-renderer-fixture-bundle-ts','component:apps-web-src-inspector-scan-one-pager-ts'].map(id=>{
const source=semanticBounds(web,id,'component'); const cold=fixture.createScene(id);const live=fixture.createScene(id,web); const center=(r:any)=>r?{x:r.x+r.width/2,y:r.y+r.height/2}:null;const c=center(source)!;
const codeCenter=center(semanticBounds(cold,id,'code'))!;const window={x:codeCenter.x-20,y:codeCenter.y-20,width:40,height:40};const keepId=snapshot.entities.find((e:any)=>e.parentId===id)?.id;const paged=fixture.createScene(id,cold,{worldBounds:window,keepEntityIds:[id,keepId]});
const coldBounds=cold.projection!.boundsByEntityIdAndDetail;const pagedBounds=paged.projection!.boundsByEntityIdAndDetail;const codeIds=cold.projection!.entityIdsByDetail.code.filter(cid=>snapshot.entities.find((e:any)=>e.id===cid)?.kind==='code');const pagedCodeIds=paged.projection!.entityIdsByDetail.code.filter(cid=>snapshot.entities.find((e:any)=>e.id===cid)?.kind==='code');const intersect=(r:any)=>r&&r.x<window.x+window.width&&r.x+r.width>window.x&&r.y<window.y+window.height&&r.y+r.height>window.y;const expectedInWindow=codeIds.filter(cid=>intersect(coldBounds[cid]?.code));const missingInWindow=expectedInWindow.filter(cid=>!pagedCodeIds.includes(cid));const sharedChanges=pagedCodeIds.filter(cid=>codeIds.includes(cid)&&JSON.stringify(coldBounds[cid]?.code)!==JSON.stringify(pagedBounds[cid]?.code));return {id,expectedInWindow,missingInWindow,sharedChanges,source,sourceCenter:c,cold:semanticBounds(cold,id,'component'),live:semanticBounds(live,id,'component'),paged:semanticBounds(paged,id,'component'),coldCode:semanticBounds(cold,id,'code'),pagedCode:semanticBounds(paged,id,'code'),window,coldComponentIds:cold.projection?.entityIdsByDetail.component.length,pagedComponentIds:paged.projection?.entityIdsByDetail.component.length,coldCodeIds:cold.projection?.entityIdsByDetail.code.length,pagedCodeIds:paged.projection?.entityIdsByDetail.code.length};});
writeFileSync('../../docs/qa/zoom-continuity/component-anchor-audit-after-pinned.json',JSON.stringify({snapshotId:snapshot.id,targetAspect:1.6,rows},null,2)+'\n');console.log(JSON.stringify(rows));
},60000);
